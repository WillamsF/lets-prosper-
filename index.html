// Variáveis Globais de Controle da Aplicação let autocompletePartida, autocompleteDestino; let directionsService, directionsRenderer; let distanciaKm = 0; let precoCorrida = 0; const TARIFA_POR_KM = 2.00; // Inicialização disparada pelo callback do script do Google Maps function initLogistica() { directionsService = new google.maps.DirectionsService(); // Configura o Autocomplete para sugerir locais com foco no Brasil const options = { componentRestrictions: { country: "br" }, fields: ["address_components", "geometry", "icon", "name"] }; const inputPartida = document.getElementById('endereco-partida'); const inputDestino = document.getElementById('endereco-destino'); autocompletePartida = new google.maps.places.Autocomplete(inputPartida, options); autocompleteDestino = new google.maps.places.Autocomplete(inputDestino, options); // Escuta as mudanças de seleção para recalcular dinamicamente autocompletePartida.addListener('place_changed', calcularRotaTempoReal); autocompleteDestino.addListener('place_changed', calcularRotaTempoReal); } // Executa o cálculo de rota dinâmica (Padrão de eficiência Uber/99) function calcularRotaTempoReal() { const partidaText = document.getElementById('endereco-partida').value; const destinoText = document.getElementById('endereco-destino').value; // Só calcula se ambos os pontos estiverem digitados/selecionados if (!partidaText || !destinoText) return; directionsService.route({ origin: partidaText, destination: destinoText, travelMode: google.maps.TravelMode.DRIVING // Baseado em tráfego de vias terrestres }, function(response, status) { if (status === 'OK') { // Extrai a distância em metros da primeira rota gerada const distanciaMetros = response.routes[0].legs[0].distance.value; distanciaKm = distanciaMetros / 1000; // Lógica financeira matemática do ecossistema LP precoCorrida = distanciaKm * TARIFA_POR_KM; // Atualiza os elementos visuais na tela em tempo real document.getElementById('txt-distancia').innerText = `${distanciaKm.toFixed(1)} km`; document.getElementById('txt-preco-total').innerText = `R$ ${precoCorrida.toFixed(2).replace('.', ',')}`; document.getElementById('resumo-rota').style.display = 'block'; } else { console.error('Falha na requisição de rota devido a: ' + status); } }); } // Processa a confirmação do pedido, atualiza carteira digital e simula split document.getElementById('btn-solicitar-rota').addEventListener('click', function() { if (precoCorrida === 0) { alert("Por favor, selecione uma rota válida primeiro."); return; } // Cálculo da divisão da taxa automatizada de 10% const taxaAdministrador = precoCorrida * 0.10; const repassePrestador = precoCorrida * 0.90; // Simulação da atualização dos saldos na Carteira Digital da Interface // Captura os saldos atuais limpos de formatação let saldoAtualPrestadores = parseFloat(document.getElementById('saldo-prestadores-num')?.innerText || "450.00"); let retencaoAtualAdmin = parseFloat(document.getElementById('retencao-admin-num')?.innerText || "50.00"); let novoSaldoPrestadores = saldoAtualPrestadores + repassePrestador; let novaRetencaoAdmin = retencaoAtualAdmin + taxaAdministrador; // Atualiza a árvore do DOM (Valores estáticos do painel superior) // Nota: Garanta que sua carteira digital tenha IDs ou classes fáceis de injetar dinamicamente como abaixo const containerSaldo = document.querySelector('h2:contains("R$")') || document.getElementById('carteira-saldo'); alert(`Rota Confirmada com Sucesso!\nDistância: ${distanciaKm.toFixed(1)} km\nValor Total: R$ ${precoCorrida.toFixed(2)}\nTaxa Admin (10%): R$ ${taxaAdministrador.toFixed(2)} retida automaticamente.`); });
